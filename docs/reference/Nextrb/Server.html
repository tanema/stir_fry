def run!(app_klass)
  return unless running_server.nil?

  Rackup::Handler.default.run(app_klass) do |server|
    at_exit { quit! }
    SIGNALS.each { |signal| chain_trap(signal) { quit! } }
    @running_server = server
  end
ensure
  quit!
end
